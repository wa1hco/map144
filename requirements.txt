# NumPy must stay < 2.x: UHD 4.6.0 (python3-uhd) was compiled against
# NumPy 1.26.4 (system package).  NumPy 2.x changes PyArrayObject layout
# and causes a seg fault in usrp_source._recv_loop.  All other deps
# (matplotlib, pyqtgraph, scipy, bottleneck) work fine with 1.26.x.
numpy>=1.26.4,<2
bottleneck>=1.3.7
matplotlib==3.10.6
PyQt5==5.15.11
pyqtgraph==0.14.0
scipy

# numba is used by the channelizer hot path for an @njit kernel.  Constrain
# it to versions that support NumPy 1.26.x — older numba (<0.59) breaks on
# NumPy 1.24+ ABI changes, and very-recent numba may require NumPy 2.x.
# 0.59–0.65 is the working range against numpy 1.26 on Python 3.12.
numba>=0.59,<0.66

# flexclient is the SmartSDR / Flex-6000 client library used by the Flex
# source module.  Installed on demand; not strictly required when running
# map144 against other source types (USRP, Airspy, RTL-SDR, WAV replay).
flexclient>=0.1
